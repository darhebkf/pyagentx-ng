# Tables

Efficiently query SNMP tables.

## Overview

SNMP tables are indexed collections of rows. Each row has the same columns, indexed by one or more values. For example, `ifTable` is indexed by `ifIndex`:

```
ifDescr.1 = "lo"
ifDescr.2 = "eth0"
ifDescr.3 = "eth1"
ifType.1 = softwareLoopback(24)
ifType.2 = ethernetCsmacd(6)
ifType.3 = ethernetCsmacd(6)
```

## Basic Table Walk

### Walking a Single Column

```python
async with Manager("192.168.1.1") as mgr:
    async for oid, value in mgr.bulk_walk("1.3.6.1.2.1.2.2.1.2"):
        idx = oid.split(".")[-1]
        print(f"Interface {idx}: {value}")
```

### Walking Multiple Columns

```python
async def get_interface_table(mgr: Manager) -> dict[str, dict]:
    table = {}
    base = "1.3.6.1.2.1.2.2.1"

    columns = {
        "ifDescr": "2",
        "ifType": "3",
        "ifMtu": "4",
        "ifSpeed": "5",
        "ifOperStatus": "8",
    }

    for col_name, col_id in columns.items():
        async for oid, value in mgr.bulk_walk(f"{base}.{col_id}"):
            idx = oid.split(".")[-1]
            if idx not in table:
                table[idx] = {}
            table[idx][col_name] = value

    return table
```

## Table Helper

### get_table

Retrieve an entire table efficiently:

```python
from snmpkit.manager import Manager

async with Manager("192.168.1.1") as mgr:
    table = await mgr.get_table(
        "1.3.6.1.2.1.2.2",  # ifTable
        columns=["1.2", "1.3", "1.8"],  # ifDescr, ifType, ifOperStatus
    )

    for idx, row in table.items():
        print(f"{idx}: {row}")
```

Output:
```python
{
    "1": {"1.2": "lo", "1.3": 24, "1.8": 1},
    "2": {"1.2": "eth0", "1.3": 6, "1.8": 1},
    "3": {"1.2": "eth1", "1.3": 6, "1.8": 2},
}
```

## Sparse Tables

Some tables have missing values (sparse tables). Handle them gracefully:

```python
table = await mgr.get_table(
    "1.3.6.1.2.1.4.22",  # ipNetToMediaTable (ARP table)
    columns=["1.2", "1.3", "1.4"],
    default=None,  # Return None for missing values
)

for idx, row in table.items():
    mac = row.get("1.2") or "unknown"
    ip = row.get("1.3") or "unknown"
    print(f"{ip} -> {mac}")
```

## Compound Indices

Some tables have multi-part indices:

### IP Route Table

```python
# ipRouteTable is indexed by ipRouteDest (4 octets)
async for oid, value in mgr.bulk_walk("1.3.6.1.2.1.4.21.1.1"):
    # OID: 1.3.6.1.2.1.4.21.1.1.10.0.0.0
    parts = oid.split(".")
    dest = ".".join(parts[-4:])  # Last 4 parts are the IP
    print(f"Route to {dest}: {value}")
```

### TCP Connection Table

```python
# tcpConnTable is indexed by (localAddr, localPort, remoteAddr, remotePort)
async for oid, value in mgr.bulk_walk("1.3.6.1.2.1.6.13.1.1"):
    parts = oid.split(".")
    local_addr = ".".join(parts[-10:-6])
    local_port = parts[-6]
    remote_addr = ".".join(parts[-5:-1])
    remote_port = parts[-1]
    print(f"{local_addr}:{local_port} -> {remote_addr}:{remote_port}: {value}")
```

### Index Parsing Helper

```python
from snmpkit.manager import parse_table_index

# Define index structure
index_def = [
    ("ifIndex", 1),       # Single integer
    ("ipAddress", 4),     # 4 octets for IP
]

async for oid, value in mgr.bulk_walk("1.3.6.1.2.1.4.20.1.1"):
    indices = parse_table_index(oid, "1.3.6.1.2.1.4.20.1.1", index_def)
    print(f"Interface {indices['ifIndex']}, IP {indices['ipAddress']}: {value}")
```

## Efficient Table Retrieval

### Parallel Column Fetching

Fetch multiple columns concurrently:

```python
import asyncio

async def get_interface_table_fast(mgr: Manager) -> dict:
    base = "1.3.6.1.2.1.2.2.1"
    columns = ["2", "3", "8", "10", "16"]  # descr, type, status, inOctets, outOctets

    async def fetch_column(col_id):
        result = []
        async for oid, value in mgr.bulk_walk(f"{base}.{col_id}"):
            idx = oid.split(".")[-1]
            result.append((idx, col_id, value))
        return result

    results = await asyncio.gather(*[fetch_column(c) for c in columns])

    table = {}
    for column_results in results:
        for idx, col_id, value in column_results:
            if idx not in table:
                table[idx] = {}
            table[idx][col_id] = value

    return table
```

### Bulk Size Tuning

Adjust `bulk_size` based on table width and network conditions:

```python
# Wide tables with many columns - smaller bulk_size
async for oid, value in mgr.bulk_walk("1.3.6.1.2.1.2.2.1.2", bulk_size=10):
    pass

# Narrow tables (single column) - larger bulk_size
async for oid, value in mgr.bulk_walk("1.3.6.1.2.1.2.2.1.2", bulk_size=50):
    pass
```

<Callout type="info">
Larger `bulk_size` means fewer round-trips but larger packets. Most devices support 25-50. Some older devices may only support 10.
</Callout>

## Example: Interface Report

```python
import asyncio
from dataclasses import dataclass
from snmpkit.manager import Manager

@dataclass
class Interface:
    index: int
    name: str
    type: int
    speed: int
    status: str
    in_octets: int
    out_octets: int
    in_errors: int
    out_errors: int

async def get_interfaces(host: str) -> list[Interface]:
    async with Manager(host) as mgr:
        table = await mgr.get_table(
            "1.3.6.1.2.1.2.2",
            columns={
                "name": "1.2",
                "type": "1.3",
                "speed": "1.5",
                "status": "1.8",
                "in_oct": "1.10",
                "out_oct": "1.16",
                "in_err": "1.14",
                "out_err": "1.20",
            },
        )

        interfaces = []
        for idx, row in table.items():
            interfaces.append(Interface(
                index=int(idx),
                name=str(row["name"]),
                type=int(row["type"]),
                speed=int(row["speed"]),
                status="up" if row["status"] == 1 else "down",
                in_octets=int(row["in_oct"]),
                out_octets=int(row["out_oct"]),
                in_errors=int(row["in_err"]),
                out_errors=int(row["out_err"]),
            ))

        return sorted(interfaces, key=lambda i: i.index)

async def main():
    interfaces = await get_interfaces("192.168.1.1")

    print(f"{'Name':<15} {'Status':<8} {'Speed':<12} {'In (MB)':<12} {'Out (MB)':<12}")
    print("-" * 60)

    for iface in interfaces:
        speed_mbps = iface.speed // 1_000_000
        in_mb = iface.in_octets / 1024 / 1024
        out_mb = iface.out_octets / 1024 / 1024
        print(f"{iface.name:<15} {iface.status:<8} {speed_mbps:<12} {in_mb:<12.1f} {out_mb:<12.1f}")

asyncio.run(main())
```
