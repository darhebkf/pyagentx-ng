# Traps & Notifications

Receive and send SNMP traps and notifications.

## Overview

SNMP devices send unsolicited messages to report events:

| Type | Version | Confirmation |
|------|---------|--------------|
| Trap (v1) | SNMPv1 | No |
| Trap (v2c/v3) | SNMPv2c, SNMPv3 | No |
| Inform | SNMPv2c, SNMPv3 | Yes (acknowledged) |

## Receiving Traps

### Basic Trap Receiver

```python
import asyncio
from snmpkit.manager import TrapReceiver

async def handle_trap(trap):
    print(f"Trap from {trap.source}")
    print(f"  Type: {trap.trap_oid}")
    print(f"  Uptime: {trap.uptime}")

    for varbind in trap.varbinds:
        print(f"  {varbind.oid} = {varbind.value}")

async def main():
    receiver = TrapReceiver(port=162)
    receiver.on_trap(handle_trap)

    print("Listening for traps on UDP port 162...")
    await receiver.start()

asyncio.run(main())
```

<Callout type="warning">
Port 162 requires root/administrator privileges. Use a higher port (e.g., 1162) for testing.
</Callout>

### Trap Object

```python
@dataclass
class Trap:
    source: tuple[str, int]    # (ip, port)
    version: int               # 1, 2, or 3
    community: str | None      # v1/v2c community
    user: str | None           # v3 username
    trap_oid: str              # Trap type OID
    enterprise: str | None     # v1 enterprise OID
    uptime: int                # sysUpTime value
    varbinds: list[VarBind]    # Variable bindings
    context: str | None        # v3 context
```

### Filtering Traps

```python
from snmpkit.manager import TrapReceiver, TrapFilter

# Filter by source
receiver = TrapReceiver(
    port=162,
    filters=[
        TrapFilter(source="192.168.1.0/24"),
    ]
)

# Filter by trap type
receiver = TrapReceiver(
    port=162,
    filters=[
        TrapFilter(trap_oid="1.3.6.1.6.3.1.1.5.3"),  # linkDown
        TrapFilter(trap_oid="1.3.6.1.6.3.1.1.5.4"),  # linkUp
    ]
)

# Filter by community
receiver = TrapReceiver(
    port=162,
    filters=[
        TrapFilter(community="monitoring"),
    ]
)
```

### Multiple Handlers

```python
async def log_trap(trap):
    print(f"[LOG] {trap.source}: {trap.trap_oid}")

async def alert_on_critical(trap):
    critical_oids = [
        "1.3.6.1.6.3.1.1.5.3",  # linkDown
        "1.3.6.1.4.1.9.9.13.3.0.1",  # ciscoEnvMonFan
    ]
    if trap.trap_oid in critical_oids:
        await send_alert(trap)

receiver = TrapReceiver(port=162)
receiver.on_trap(log_trap)
receiver.on_trap(alert_on_critical)
```

## SNMPv3 Traps

### Configuring v3 Security

```python
from snmpkit.manager import TrapReceiver, USMUser, AuthProtocol, PrivProtocol

users = [
    USMUser(
        username="trapuser",
        auth_protocol=AuthProtocol.SHA256,
        auth_key="auth_password",
        priv_protocol=PrivProtocol.AES128,
        priv_key="priv_password",
    ),
]

receiver = TrapReceiver(port=162, users=users)
```

## Sending Traps

### Send Trap (v2c)

```python
from snmpkit.manager import TrapSender
from snmpkit.core import Oid, Value, VarBind

async def main():
    sender = TrapSender(
        destination="192.168.1.100",
        port=162,
        community="public",
    )

    await sender.send_trap(
        trap_oid="1.3.6.1.4.1.12345.0.1",
        varbinds=[
            VarBind(Oid("1.3.6.1.4.1.12345.1.1.0"), Value.Integer(95)),
            VarBind(Oid("1.3.6.1.4.1.12345.1.2.0"), Value.OctetString(b"CPU High")),
        ],
    )

asyncio.run(main())
```

### Send Inform (Acknowledged)

```python
from snmpkit.manager import TrapSender

async def main():
    sender = TrapSender(
        destination="192.168.1.100",
        port=162,
        community="public",
    )

    try:
        await sender.send_inform(
            trap_oid="1.3.6.1.4.1.12345.0.1",
            varbinds=[...],
            timeout=5.0,
        )
        print("Inform acknowledged")
    except TimeoutError:
        print("No acknowledgment received")

asyncio.run(main())
```

<Callout type="info">
Inform requests wait for acknowledgment. Use traps when acknowledgment isn't needed.
</Callout>

## Common Trap OIDs

### Standard Traps

| OID | Name | Description |
|-----|------|-------------|
| `1.3.6.1.6.3.1.1.5.1` | coldStart | Agent reinitializing |
| `1.3.6.1.6.3.1.1.5.2` | warmStart | Agent reinitializing (config preserved) |
| `1.3.6.1.6.3.1.1.5.3` | linkDown | Interface went down |
| `1.3.6.1.6.3.1.1.5.4` | linkUp | Interface came up |
| `1.3.6.1.6.3.1.1.5.5` | authenticationFailure | Bad community/credentials |

### Decoding Trap OIDs

```python
from snmpkit.mibs import MibRegistry

registry = MibRegistry()

async def handle_trap(trap):
    trap_name = registry.lookup(trap.trap_oid)
    print(f"Received: {trap_name}")

    if trap.trap_oid == "1.3.6.1.6.3.1.1.5.3":  # linkDown
        # Extract ifIndex from varbinds
        for vb in trap.varbinds:
            if "ifIndex" in registry.lookup(vb.oid):
                print(f"Interface {vb.value} went down")
```

import asyncio
import sqlite3
import json
from datetime import datetime
from snmpkit.manager import TrapReceiver

conn = sqlite3.connect("traps.db")
conn.execute("""
    CREATE TABLE IF NOT EXISTS traps (
        id INTEGER PRIMARY KEY,
        timestamp TEXT,
        source TEXT,
        trap_oid TEXT,
        varbinds TEXT
    )
""")

async def log_to_db(trap):
    conn.execute(
        "INSERT INTO traps (timestamp, source, trap_oid, varbinds) VALUES (?, ?, ?, ?)",
        (
            datetime.now().isoformat(),
            f"{trap.source[0]}:{trap.source[1]}",
            trap.trap_oid,
            json.dumps([(str(vb.oid), str(vb.value)) for vb in trap.varbinds]),
        ),
    )
    conn.commit()

async def main():
    receiver = TrapReceiver(port=1162)
    receiver.on_trap(log_to_db)
    await receiver.start()

asyncio.run(main())

## Example: Alert Forwarder

```python
import asyncio
import httpx
from snmpkit.manager import TrapReceiver

CRITICAL_TRAPS = {
    "1.3.6.1.6.3.1.1.5.3": "Link Down",
    "1.3.6.1.4.1.9.9.13.3.0.1": "Fan Failure",
    "1.3.6.1.4.1.9.9.13.3.0.3": "Temperature Alert",
}

WEBHOOK_URL = "https://hooks.slack.com/services/xxx/yyy/zzz"

async def forward_critical(trap):
    alert_name = CRITICAL_TRAPS.get(trap.trap_oid)
    if not alert_name:
        return

    message = {
        "text": f":warning: *{alert_name}* from {trap.source[0]}",
        "attachments": [
            {
                "fields": [
                    {"title": str(vb.oid), "value": str(vb.value), "short": True}
                    for vb in trap.varbinds[:10]
                ],
            }
        ],
    }

    async with httpx.AsyncClient() as client:
        await client.post(WEBHOOK_URL, json=message)

async def main():
    receiver = TrapReceiver(port=162)
    receiver.on_trap(forward_critical)
    await receiver.start()

asyncio.run(main())
```
